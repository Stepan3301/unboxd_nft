        // Products for the roulette with NFT-style items
        const products = [

                stars: 1
            },
            { 
                name: "Plasma Rifle", 
                image: "https://placehold.co/200x200/ff5500/111111?text=Plasma+Rifle", 
                rarity: "legendary",
                stars: 5
            },
            { 
                name: "Techno Mask", 
                image: "https://placehold.co/200x200/00ffaa/111111?text=Techno+Mask", 
                rarity: "rare",
                stars: 3
            }
        ];

        // Create roulette items
        const roulette = document.getElementById('roulette');
        const itemWidth = 200; // Width of each item in pixels
        
        // Create multiple copies of items to make the roulette longer
        const totalItems = products.length * 3; // 3 copies of each item
        const rouletteWidth = totalItems * itemWidth;
        roulette.style.width = rouletteWidth + 'px';
        
        for (let i = 0; i < 3; i++) {
            products.forEach((product, index) => {
                const item = document.createElement('div');
                item.className = 'item';
                
                const content = document.createElement('div');
                content.className = 'item-content';
                
                const img = document.createElement('img');
                img.src = product.image;
                img.alt = product.name;
                
                const name = document.createElement('div');
                name.className = 'item-name';
                name.textContent = product.name;
                
                const rarity = document.createElement('div');
                rarity.className = `item-rarity ${product.rarity}`;
                rarity.textContent = product.rarity.toUpperCase();
                
                content.appendChild(img);
                content.appendChild(name);
                content.appendChild(rarity);
                item.appendChild(content);
                roulette.appendChild(item);
            });
        }

        // Roulette functionality
        const openBtn = document.getElementById('openBtn');
        const resultDiv = document.getElementById('result');
        const prizeImage = document.getElementById('prizeImage');
        const prizeName = document.getElementById('prizeName');
        const prizeRarity = document.getElementById('prizeRarity');
        const starRating = document.getElementById('starRating');
        const glitchEffect = document.querySelector('.glitch-effect');
        const loadingText = document.getElementById('loadingText');
        
        // Initial position - center the roulette
        const initialOffset = (rouletteWidth / 2) - (window.innerWidth / 2);
        roulette.style.transform = `translateX(-${initialOffset}px)`;
        
        openBtn.addEventListener('click', () => {
            // Disable button during spin
            openBtn.disabled = true;
            resultDiv.style.display = 'none';
            loadingText.style.display = 'block';
            loadingText.textContent = 'CONNECTING TO BLOCKCHAIN...';
            loadingText.style.animation = 'blink 1s infinite';
            
            // Trigger glitch effect
            glitchEffect.style.animation = 'glitch 2s';
            
            // Get current position
            const currentTransform = window.getComputedStyle(roulette).getPropertyValue('transform');
            const matrix = new DOMMatrix(currentTransform);
            const currentX = matrix.m41;
            
            // Random product selection
            const randomIndex = Math.floor(Math.random() * products.length);
            const winner = products[randomIndex];
            
            // Calculate final position
            const centerPosition = window.innerWidth / 2;
            const itemPosition = (randomIndex + products.length) * itemWidth; // Use the middle set of items
            const finalOffset = itemPosition - centerPosition + (itemWidth / 2);
            
            // Add some randomness to make it look more natural
            const randomOffset = Math.random() * 10 - 5;
            const targetX = -(finalOffset + randomOffset);
            
            // Animation duration
            const duration = 7000; // 7 seconds
            
            // Start time of animation
            const startTime = performance.now();
            // Starting position
            const startX = currentX;
            
            // Always move left (negative direction)
            // We want to ensure we're always moving in the same direction (left)
            // and that the items cycle properly
            
            // Calculate how far we need to move to reach the target
            let totalDistance;
            
            // If target is to the right of current position, we need to go left (negative)
            // by wrapping around the entire roulette
            if (targetX > startX) {
                totalDistance = targetX - startX - rouletteWidth;
            } else {
                // If target is already to the left, just go directly there
                totalDistance = targetX - startX;
            }
            
            // Ensure we move at least 3000px for a good spin effect
            if (Math.abs(totalDistance) < 3000) {
                // Add additional full rotations to ensure minimum distance
                const additionalRotations = Math.ceil((3000 - Math.abs(totalDistance)) / rouletteWidth);
                totalDistance -= additionalRotations * rouletteWidth;
            }
            
            // Animation function
            function animateSpin(currentTime) {
                // Calculate elapsed time
                const elapsedTime = currentTime - startTime;
                
                if (elapsedTime < duration) {
                    // Calculate progress (0 to 1)
                    const progress = elapsedTime / duration;
                    
                    // Easing function - starts fast, slows down (ease-out)
                    // Using cubic ease-out for smooth deceleration
                    const easedProgress = 1 - Math.pow(1 - progress, 3);
                    
                    // Calculate current position
                    let currentPosition = startX + (totalDistance * easedProgress);
                    
                    // Handle cycling of the roulette
                    // If we've moved past the end of the roulette, wrap around
                    // This ensures the items are always visible during scrolling
                    const normalizedPosition = currentPosition % rouletteWidth;
                    
                    // Apply the new position
                    roulette.style.transform = `translateX(${normalizedPosition}px)`;
                    
                    // Continue animation
                    requestAnimationFrame(animateSpin);
                    
                    // Update loading text during animation
                    if (elapsedTime > 2000 && elapsedTime < 3000) {
                        loadingText.textContent = 'VERIFYING TRANSACTION...';
                    } else if (elapsedTime > 3000) {
                        loadingText.textContent = 'GENERATING NFT...';
                    }
                } else {
                    // Animation complete - ensure we're exactly at target position
                    roulette.style.transform = `translateX(${targetX}px)`;
                    
                    // Show result
                    setTimeout(() => {
                        // Prepare result display
                        prizeImage.innerHTML = '';
                        starRating.innerHTML = '';
                        
                        const img = document.createElement('img');
                        img.src = winner.image;
                        img.alt = winner.name;
                        prizeImage.appendChild(img);
                        
                        prizeName.textContent = winner.name;
                        
                        const rarityEl = document.createElement('div');
                        rarityEl.className = `item-rarity ${winner.rarity}`;
                        rarityEl.textContent = winner.rarity.toUpperCase();
                        prizeRarity.innerHTML = '';
                        prizeRarity.appendChild(rarityEl);
                        
                        // Add star rating
                        for (let i = 0; i < winner.stars; i++) {
                            const star = document.createElement('div');
                            star.className = 'star';
                            star.textContent = '★';
                            starRating.appendChild(star);
                        }
                        
                        resultDiv.style.display = 'block';
                        openBtn.disabled = false;
                        loadingText.style.display = 'none';
                        
                        // Create confetti effect
                        createConfetti(winner.rarity);
                    }, 500);
                }
            }
            
            // Start the animation
            requestAnimationFrame(animateSpin);
        });
        
        // Confetti effect
        function createConfetti(rarity) {
            const container = document.querySelector('.container');
            const colors = {
                legendary: ['#ffaa00', '#ff5500', '#ffcc00'],
                epic: ['#aa00ff', '#ff00aa', '#cc00ff'],
                rare: ['#0088ff', '#00ffaa', '#00ccff'],
                uncommon: ['#00ff88', '#88ff00', '#aaff00'],
                common: ['#aaaaaa', '#ffffff', '#cccccc']
            };
            
            const particleCount = rarity === 'legendary' ? 150 : 
                                rarity === 'epic' ? 100 :
                                rarity === 'rare' ? 70 :
                                rarity === 'uncommon' ? 40 : 20;
            
            for (let i = 0; i < particleCount; i++) {
                const confetti = document.createElement('div');
                confetti.className = 'confetti';
                
                // Random position, color, and animation
                confetti.style.left = Math.random() * 100 + 'vw';
                const colorArray = colors[rarity] || colors.common;
                confetti.style.backgroundColor = colorArray[Math.floor(Math.random() * colorArray.length)];
                confetti.style.animation = `fall ${Math.random() * 3 + 2}s linear forwards`;
                confetti.style.opacity = '1';
                confetti.style.transform = `translateY(0) rotate(${Math.random() * 360}deg)`;
                confetti.style.width = `${Math.random() * 10 + 5}px`;
                confetti.style.height = `${Math.random() * 10 + 5}px`;
                
                // Add keyframe animation dynamically
                const keyframes = `
                    @keyframes fall {
                        to {
                            transform: translateY(100vh) rotate(${Math.random() * 360 + 360}deg);
                            opacity: 0;
                        }
                    }
                `;
                
                const style = document.createElement('style');
                style.innerHTML = keyframes;
                document.head.appendChild(style);
                
                container.appendChild(confetti);
                
                // Remove confetti after animation
                setTimeout(() => {
                    confetti.remove();
                    style.remove();
                }, 5000);
            }
        }
